% --- Валидация конфигурации билета (datatool already loaded above) ---
% 1) NumberOfQuestions: шаблон поддерживает только 2 или 3 вопроса в билете.
\ifnum\NumberOfQuestions<2\relax
  \PackageError{exam-cards}%
    {NumberOfQuestions=\NumberOfQuestions, but only 2 or 3 are supported}%
    {Set \string\NumberOfQuestions to 2 or 3 in config.tex.}%
\else\ifnum\NumberOfQuestions>3\relax
  \PackageError{exam-cards}%
    {NumberOfQuestions=\NumberOfQuestions, but only 2 or 3 are supported}%
    {Set \string\NumberOfQuestions to 2 or 3 in config.tex.}%
\fi\fi
% 2) Число столбцов в CSV должно совпадать с NumberOfQuestions.
\ifnum\DTLcolumncount{questions}=\NumberOfQuestions\relax\else
  \PackageError{exam-cards}%
    {CSV file '\FileQuestions' has \DTLcolumncount{questions} columns but NumberOfQuestions=\NumberOfQuestions}%
    {The number of columns in \FileQuestions\space must match \string\NumberOfQuestions\space (2 or 3).
     Fix the CSV header/rows or the \string\NumberOfQuestions\space value in config.tex.}%
\fi
