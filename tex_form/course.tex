% --- Загрузка вопросов (datatool подключён в config.tex) ---
\DTLloaddb[]{questions}{\FileQuestions}

% --- Валидация конфигурации ---
% NumberOfQuestions = число столбцов вопросов в CSV (= максимум вопросов в билете).
% В отдельном билете вопросов может быть меньше: пустая ячейка CSV пропускается.
% 1) NumberOfQuestions: шаблон поддерживает 1, 2 или 3 вопроса в билете.
\ifnum\NumberOfQuestions<1\relax
  \PackageError{exam-cards}%
    {NumberOfQuestions=\NumberOfQuestions, but only 1, 2 or 3 are supported}%
    {Set \string\NumberOfQuestions to 1, 2 or 3 in course.tex.}%
\else\ifnum\NumberOfQuestions>3\relax
  \PackageError{exam-cards}%
    {NumberOfQuestions=\NumberOfQuestions, but only 1, 2 or 3 are supported}%
    {Set \string\NumberOfQuestions to 1, 2 or 3 in course.tex.}%
\fi\fi
% 2) Число столбцов в CSV должно совпадать с NumberOfQuestions.
\ifnum\DTLcolumncount{questions}=\NumberOfQuestions\relax\else
  \PackageError{exam-cards}%
    {CSV file '\FileQuestions' has \DTLcolumncount{questions} columns but NumberOfQuestions=\NumberOfQuestions}%
    {The number of columns in \FileQuestions\space must match \string\NumberOfQuestions\space (1, 2 or 3).
     Fix the CSV header/rows or the \string\NumberOfQuestions\space value in course.tex.}%
\fi
